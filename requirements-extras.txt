# ===========================================================================
# BioHuman3D - OPTIONAL extras (voiceover + audio playback)
#
# Installed on a best-effort basis. If any of these fail, the application still
# launches and runs; it simply reports voiceover as unavailable.
#
# NOTE on the audio playback engine:
#   The upstream `pygame` project has NO wheels for Python 3.14, so pip falls
#   back to a source build that fails (Python 3.14 removed `distutils.msvccompiler`).
#   `pygame-ce` is the community-maintained drop-in fork: identical `import pygame`
#   API, and it publishes wheels for 3.12 / 3.13 / 3.14. Use it.
#
#   If you are on Python <= 3.13 and specifically want upstream pygame, swap the
#   line below for:  pygame>=2.5
# ===========================================================================

edge-tts>=6.1          # natural Microsoft neural voices, 50+ languages, no key (needs internet)
pyttsx3>=2.90          # offline SAPI5 voices (fallback when offline)
pygame-ce>=2.5         # playback of cached cloud (ElevenLabs / pre-rendered) clips

# Narrated tour video export (MP4). VTK's pip wheel does NOT include

# imageio-ffmpeg ships py3-none-win_amd64 wheels, so this works on 3.10-3.14.
